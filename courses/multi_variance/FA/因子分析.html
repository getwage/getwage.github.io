# 生成模拟数据（假设5个变量，3个公共因子）
np.random.seed(42)
n_samples = 200
data = pd.DataFrame({
    'X1': np.random.normal(0, 1, n_samples) * 0.2 + np.random.normal(0, 1, n_samples) * 0.8,
    'X2': np.random.normal(0, 1, n_samples) * 0.3 + np.random.normal(0, 1, n_samples) * 0.7,
    'X3': np.random.normal(0, 1, n_samples) * 0.4 + np.random.normal(0, 1, n_samples) * 0.6,
    'X4': np.random.normal(0, 1, n_samples) * 0.5 + np.random.normal(0, 1, n_samples) * 0.5,
    'X5': np.random.normal(0, 1, n_samples) * 0.6 + np.random.normal(0, 1, n_samples) * 0.4,
})

# 标准化数据
data = (data - data.mean()) / data.std()

# 适用性检验
kmo_all, kmo_model = calculate_kmo(data)
chi_sq, p_value = calculate_bartlett_sphericity(data)
print(f"KMO值: {kmo_model:.2f}, Bartlett检验p值: {p_value:.4f}")

# 确定因子数（特征值>1）
fa = FactorAnalyzer(n_factors=data.shape[1], rotation=None)
fa.fit(data)
ev, _ = fa.get_eigenvalues()
plt.scatter(range(1, data.shape[1]+1), ev)
plt.plot(range(1, data.shape[1]+1), ev)
plt.title('碎石图')
plt.xlabel('因子数')
plt.ylabel('特征值')
plt.axhline(y=1, color='r', linestyle='--')
plt.show()

KMO值: 0.51, Bartlett检验p值: 0.5643
